cmake_minimum_required(VERSION 3.20)
project(abcurves_reference_comparison LANGUAGES C CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ABC_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ABC_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(ABC_PROFILE OFF CACHE BOOL "" FORCE)
add_subdirectory(../../lib engine)
add_executable(reference_benchmark native.cpp)
target_link_libraries(reference_benchmark PRIVATE abcurves::native)
target_include_directories(reference_benchmark PRIVATE ../../lib/src/core)
if(MSVC)
  target_compile_options(reference_benchmark PRIVATE /O2 /fp:strict /permissive-)
else()
  target_compile_options(reference_benchmark PRIVATE -O3 -fno-fast-math -ffp-contract=off)
endif()
