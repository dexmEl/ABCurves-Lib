# Keep optional development targets out of library-only builds.
function(abc_c_program name source)
 add_executable(${name} ${source})
 target_link_libraries(${name} PRIVATE abcurves)
 set_target_properties(${name} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
endfunction()

option(ABC_ENABLE_ASAN "Build API/native validation with AddressSanitizer" OFF)
if(ABC_ENABLE_ASAN)
 foreach(target abcurves abcurves_native abcurves_renderer_core abc_c_contract abc_cpp_contract abc_c_renderer abc_c_observation abcurves_observation_contract abc_c_segments abcurves_segment_contract abc_c_recovery abc_api_hardening abc_c_hardening)

   if(MSVC)
    target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO)
   else()
    target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address)
   endif()
  endif()
 endforeach()
